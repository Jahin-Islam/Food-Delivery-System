-- TRIGGER 1: Auto-update restaurant rating when a review is INSERTED
DELIMITER $$

CREATE TRIGGER trg_update_rating_on_insert
AFTER INSERT ON reviews_review
FOR EACH ROW
BEGIN
    UPDATE resturants_restaurant res
    INNER JOIN (
        SELECT
            o.restaurant_id,
            AVG(r.rating)  AS avg_rating,
            COUNT(r.id)    AS total_rated
        FROM reviews_review r
        INNER JOIN orders_order o ON o.order_id = r.order_id
        WHERE o.restaurant_id = (
            SELECT restaurant_id FROM orders_order WHERE order_id = NEW.order_id
        )
        GROUP BY o.restaurant_id
    ) agg ON agg.restaurant_id = res.id
    SET res.rating      = ROUND(agg.avg_rating, 2),
        res.total_rated = agg.total_rated
    WHERE res.id = agg.restaurant_id;
END$$

DELIMITER ;

CREATE TRIGGER trg_nullify_order_address_on_delete
BEFORE DELETE ON addresses_deliveryaddress
FOR EACH ROW
BEGIN
    UPDATE orders_order
    SET address_id = NULL
    WHERE address_id = OLD.id;
END$$

CREATE TRIGGER trg_set_delivered_at
BEFORE UPDATE ON orders_order
FOR EACH ROW
BEGIN
    IF NEW.status = 'DELIVERED' AND OLD.status != 'DELIVERED' THEN
        SET NEW.delivered_at = NOW();
    END IF;
END$$

-- FUNCTION: Calculate rider's earnings for today
DELIMITER $$

CREATE FUNCTION fn_rider_today_earnings(p_rider_id INT)
RETURNS DECIMAL(10,2)
DETERMINISTIC
READS SQL DATA
BEGIN
    DECLARE v_earnings DECIMAL(10,2);

    SELECT COALESCE(
        SUM((COALESCE(delivery_charge, 0) + COALESCE(rider_tip, 0)) * 0.50),
        0.00
    )
    INTO v_earnings
    FROM orders_order
    WHERE rider_id  = p_rider_id
      AND status    = 'DELIVERED'
      AND DATE(delivered_at) = CURDATE();

    RETURN v_earnings;
END$$

DELIMITER ;

-- FUNCTION: Calculate total delivered orders for a rider today
DELIMITER $$

CREATE FUNCTION fn_rider_today_order_count(p_rider_id INT)
RETURNS INT
DETERMINISTIC
READS SQL DATA
BEGIN
    DECLARE v_count INT;

    SELECT COUNT(*)
    INTO v_count
    FROM orders_order
    WHERE rider_id  = p_rider_id
      AND status    = 'DELIVERED'
      AND DATE(delivered_at) = CURDATE();

    RETURN v_count;
END$$

DELIMITER ;

DELIMITER $$
CREATE FUNCTION fn_get_customer_id(p_user_id INT)
RETURNS INT
READS SQL DATA
DETERMINISTIC
BEGIN
    DECLARE v_id INT DEFAULT NULL;
    SELECT id INTO v_id
    FROM customers_customer
    WHERE user_id = p_user_id
    LIMIT 1;
    RETURN v_id;
END$$

-- PROCEDURE: Place a new order atomically
DELIMITER $$

CREATE PROCEDURE sp_place_order(
    IN  p_restaurant_id   INT,
    IN  p_customer_id     INT,
    IN  p_address_id      INT,
    IN  p_total_amount    DECIMAL(10,2),
    IN  p_discount_amount DECIMAL(10,2),
    IN  p_delivery_charge DECIMAL(10,2),
    IN  p_service_charge  DECIMAL(10,2),
    IN  p_rider_tip       DECIMAL(10,2),
    IN  p_email           VARCHAR(254),
    IN  p_first_name      VARCHAR(50),
    IN  p_last_name       VARCHAR(50),
    IN  p_phone_number    VARCHAR(50),
    OUT p_order_id        INT
)
BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    START TRANSACTION;

    INSERT INTO orders_order
        (restaurant_id, customer_id, address_id, status,
         total_amount, discount_amount, delivery_charge,
         service_charge, rider_tip,
         email, first_name, last_name, phone_number, created_at)
    VALUES
        (p_restaurant_id, p_customer_id, p_address_id, 'PENDING',
         p_total_amount, p_discount_amount, p_delivery_charge,
         p_service_charge, p_rider_tip,
         p_email, p_first_name, p_last_name, p_phone_number, NOW());

    SET p_order_id = LAST_INSERT_ID();

    COMMIT;
END$$

DELIMITER ;

DELIMITER $$

CREATE PROCEDURE sp_cancel_order(
    IN  p_order_id   INT,
    OUT p_success    TINYINT,
    OUT p_message    VARCHAR(200)
)
BEGIN
    DECLARE v_current_status VARCHAR(20);

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SET p_success = 0;
        SET p_message = 'Unexpected error during cancellation.';
    END;

    START TRANSACTION;

    SELECT status INTO v_current_status
    FROM orders_order
    WHERE order_id = p_order_id
    FOR UPDATE;

    IF v_current_status IS NULL THEN
        ROLLBACK;
        SET p_success = 0;
        SET p_message = 'Order not found.';

    ELSEIF v_current_status IN ('DELIVERED', 'CANCELLED') THEN
        ROLLBACK;
        SET p_success = 0;
        SET p_message = CONCAT('Cannot cancel an order that is already ', v_current_status, '.');

    ELSE
        UPDATE orders_order
        SET status   = 'CANCELLED',
            rider_id = NULL
        WHERE order_id = p_order_id;

        COMMIT;
        SET p_success = 1;
        SET p_message = 'Order cancelled successfully.';
    END IF;
END$$

DELIMITER ;
